\usepackage[many]{tcolorbox}

% Estilo para Historias de Usuario
\newtcolorbox{UserStoryBox}[2]{%
    enhanced,
    colback=BgSoftGray,
    colframe=PrimaryNavy,
    coltitle=white,
    fonttitle=\bfseries\sffamily,
    title={#1 \hfill \small\textbf{ID: #2}},
    arc=3mm,
    boxrule=1pt,
    titlerule=0pt,
    top=8pt,
    bottom=8pt,
    left=12pt,
    right=12pt,
    drop shadow=black!8!white,
    before skip=12pt,
    after skip=12pt
}
